%--- Vai trò trong hệ thống ---
\begin{frame}{Các vai trò trong hệ thống}
\begin{itemize}
  \item \textbf{Khách hàng (Customer)}: Tìm kiếm, lọc voucher, quản lý giỏ hàng, thanh toán trực tuyến, quản lý kho voucher cá nhân, đánh giá \& bình luận.
  \item \textbf{Đối tác (Partner)}: Đăng ký doanh nghiệp, quản lý danh sách chi nhánh, quản lý nhân viên, tạo và phát hành các chương trình voucher, xem báo cáo doanh thu.
  \item \textbf{Nhân viên đối tác (Partner Employee)}: Phụ trách từng chi nhánh, sử dụng camera quét mã QR / Barcode để tra cứu và khấu trừ (Redeem) voucher tại quầy.
  \item \textbf{Quản trị viên (Admin)}: Kiểm duyệt đối tác, quản lý người dùng, quản lý đơn hàng \& voucher toàn sàn, quản trị nội dung CMS (Category, Banner, Popup, Bài viết), xem nhật ký hệ thống.
\end{itemize}
\end{frame}

%--- Công nghệ và stack ---
\begin{frame}{Công nghệ \& Công cụ phát triển}
\begin{columns}
  \column{0.5\textwidth}
  \textbf{Frontend}
  \begin{itemize}
    \item \textbf{Framework}: Next.js 16 (React 18, SSR \& App Router)
    \item \textbf{Styling}: Tailwind CSS, Lucide Icons, Sonner toast
    \item \textbf{Tiện ích QR}: \texttt{qrcode.react}, \texttt{@zxing/browser} (quét QR từ Camera)
    \item \textbf{UI Drag-Drop}: \texttt{@dnd-kit} (sắp xếp CMS)
  \end{itemize}
  \column{0.5\textwidth}
  \textbf{Backend \& Hạ tầng}
  \begin{itemize}
    \item \textbf{Runtime}: Node.js (TypeScript, ES Modules)
    \item \textbf{Framework}: Express 5
    \item \textbf{Database}: PostgreSQL (kết nối qua Connection Pool)
    \item \textbf{Lưu trữ media}: AWS S3 Bucket
    \item \textbf{Bảo mật}: JWT, Bcrypt, Helmet, CORS, Rate Limit
    \item \textbf{Email}: Nodemailer (SMTP OTP \& thông báo)
  \end{itemize}
\end{columns}
\end{frame}

%--- Đa dạng Cổng thanh toán ---
\begin{frame}{Cổng thanh toán tích hợp (Multi-Payment Gateways)}
\begin{itemize}
  \item Hỗ trợ đa dạng phương thức thanh toán trong nước và quốc tế:
    \begin{itemize}
      \item \textbf{Stripe}: Thanh toán thẻ quốc tế Visa / MasterCard / JCB (hỗ trợ Stripe Elements \& Webhook).
      \item \textbf{PayPal}: Thanh toán qua cổng thanh toán quốc tế PayPal SDK / Capture Order.
      \item \textbf{VNPay}: Cổng thanh toán nội địa qua VNPay Sandbox (ATM nội địa / Internet Banking / VNPAY-QR).
      \item \textbf{ZaloPay}: Cổng thanh toán ví điện tử ZaloPay Sandbox (quét mã QR / Gateway Redirect).
    \end{itemize}
  \item \textbf{Quy trình xử lý}: Tạo đơn hàng $\rightarrow$ Khởi tạo phiên thanh toán $\rightarrow$ Xử lý Redirect/Capture/Webhook $\rightarrow$ Cập nhật trạng thái \& tự động phát hành mã E-Voucher.
\end{itemize}
\end{frame}

%--- Tính năng nổi bật ---
\begin{frame}{Tính năng nổi bật của đồ án}
\begin{itemize}
  \item \textbf{Quy trình Redeem Voucher hoàn chỉnh}:
    \begin{itemize}
      \item Khách hàng mở mã QR/Barcode của voucher trong kho cá nhân.
      \item Nhân viên chi nhánh dùng camera thiết bị quét mã để kiểm tra tính hợp lệ và thực hiện khấu trừ trực tiếp.
    \end{itemize}
  \item \textbf{Quản lý chi nhánh \& phân quyền nhân viên}:
    \begin{itemize}
      \item Đối tác tạo nhiều chi nhánh và gán nhân viên phụ trách từng chi nhánh cụ thể.
    \end{itemize}
  \item \textbf{Hệ thống Quản trị nội dung CMS linh hoạt}:
    \begin{itemize}
      \item Admin quản lý và kéo-thả sắp xếp Danh mục (Categories), Banners, Popups quảng cáo, Bài viết tin tức (Articles).
    \end{itemize}
  \item \textbf{Đánh giá \& Đơn hàng}: Đánh giá sao, bình luận voucher, quản lý lịch sử đơn hàng chi tiết.
\end{itemize}
\end{frame}

%--- Thông tin hosting / VPS ---
\begin{frame}{Môi trường triển khai \& Vận hành (Deployment)}
\begin{itemize}
  \item \textbf{Frontend}: Next.js Standalone Build phục vụ qua Node.js server hoặc nền tảng Vercel / Cloud.
  \item \textbf{Backend API}: Triển khai trên máy chủ ảo \textbf{VPS Linux (Ubuntu)}, quản lý tiến trình bằng \textbf{PM2} / \textbf{Docker}.
  \item \textbf{Web Server \& Proxy}: \textbf{Nginx Reverse Proxy} định tuyến traffic, cân bằng tải, nén Gzip và bảo mật SSL/TLS (Let's Encrypt / Certbot).
  \item \textbf{Cơ sở dữ liệu}: PostgreSQL Server có sao lưu định kỳ, tối ưu hóa qua Connection Pooling.
  \item \textbf{Lưu trữ tệp tin tĩnh}: Phân phối hình ảnh Banner, Voucher qua dịch vụ đám mây \textbf{AWS S3}.
\end{itemize}
\end{frame}
