\begin{frame}{Các vai trò trong hệ thống}
\begin{itemize}
  \item \textbf{Khách hàng (Customer)}: Tìm kiếm, mua và thanh toán voucher; quản lý voucher cá nhân, đánh giá \& bình luận.
  \item \textbf{Đối tác (Partner)}: Quản lý doanh nghiệp, chi nhánh, nhân viên, chương trình voucher và báo cáo doanh thu.
  \item \textbf{Nhân viên đối tác (Partner Employee)}: Phụ trách chi nhánh, quét mã QR / Barcode và khấu trừ voucher tại quầy.
  \item \textbf{Quản trị viên (Admin)}: Kiểm duyệt đối tác; quản lý người dùng, đơn hàng, voucher, nội dung CMS và nhật ký hệ thống.
\end{itemize}
\end{frame}

%--- Công nghệ và stack ---
\begin{frame}{Công nghệ \& Công cụ phát triển}
\begin{columns}
  \column{0.5\textwidth}
  \textbf{Frontend}
  \begin{itemize}
    \item \textbf{Framework}: Next.js 16 (React 18, SSR \& App Router)
    \item \textbf{Styling}: Tailwind CSS, Lucide Icons, Sonner toast
    \item \textbf{Tiện ích QR}: \texttt{qrcode.react}, \texttt{@zxing/browser} (quét QR từ Camera)
    \item \textbf{UI Drag-Drop}: \texttt{@dnd-kit} (sắp xếp CMS)
  \end{itemize}
  \column{0.5\textwidth}
  \textbf{Backend \& Hạ tầng}
  \begin{itemize}
    \item \textbf{Runtime}: Node.js (TypeScript, ES Modules)
    \item \textbf{Framework}: Express 5
    \item \textbf{Database}: Neon Database (PostgreSQL)
    \item \textbf{Lưu trữ media}: Cloudflare R2 Storage
    \item \textbf{Bảo mật}: JWT, Bcrypt, Helmet, CORS, Rate Limit
    \item \textbf{Email}: Nodemailer (SMTP OTP \& thông báo)
  \end{itemize}
\end{columns}
\end{frame}

%--- Tính năng nổi bật ---
\begin{frame}{Tính năng nổi bật của đồ án}
\small
\begin{itemize}
  \item \textbf{Quy trình Redeem Voucher hoàn chỉnh}:
    \begin{itemize}
      \item Quét QR / Barcode để kiểm tra và khấu trừ voucher tại quầy.
      \item \textbf{Giá trị}: Thanh toán nhanh hơn, hạn chế gian lận và sử dụng lại voucher.
    \end{itemize}
  \item \textbf{Quản lý chi nhánh \& phân quyền nhân viên}:
    \begin{itemize}
      \item Gán nhân viên phụ trách từng chi nhánh với quyền hạn rõ ràng.
      \item \textbf{Giá trị}: Dễ kiểm soát vận hành và truy vết giao dịch theo chi nhánh.
    \end{itemize}
  \item \textbf{Hệ thống Quản trị nội dung CMS linh hoạt}:
    \begin{itemize}
      \item Kéo-thả sắp xếp danh mục, banner, popup và bài viết.
      \item \textbf{Giá trị}: Cập nhật nội dung, chiến dịch nhanh chóng mà không cần sửa mã nguồn.
    \end{itemize}
  \item \textbf{Đánh giá \& quản lý đơn hàng}:
    \begin{itemize}
      \item Đánh giá sao, bình luận và theo dõi lịch sử đơn hàng.
      \item \textbf{Giá trị}: Tăng tính minh bạch, độ tin cậy và chất lượng hỗ trợ khách hàng.
    \end{itemize}
\end{itemize}
\end{frame}

%--- Thông tin hosting / VPS ---
\begin{frame}{Môi trường triển khai \& Vận hành (Deployment)}
\begin{itemize}
  \item \textbf{Nhà cung cấp hosting}: Oracle Cloud Infrastructure.
  \item \textbf{Cấu hình máy chủ}: 1 OCPU, bộ nhớ 1 GB và địa chỉ \texttt{168{.}107{.}65{.}236}.
  \item \textbf{Frontend}: Next.js Standalone Build phục vụ qua Node.js trên máy chủ Oracle Cloud.
  \item \textbf{Backend API}: Triển khai trên máy chủ ảo \textbf{VPS Linux (Ubuntu)}, quản lý tiến trình bằng \textbf{PM2} / \textbf{Docker}.
  \item \textbf{Web Server \& Proxy}: \textbf{Nginx Reverse Proxy} định tuyến traffic, cân bằng tải, nén Gzip và bảo mật SSL/TLS (Let's Encrypt / Certbot).
  \item \textbf{Cơ sở dữ liệu}: Neon Database (PostgreSQL), kết nối và tối ưu hóa qua Connection Pooling.
  \item \textbf{Lưu trữ tệp tin tĩnh}: Phân phối hình ảnh Banner và Voucher qua \textbf{Cloudflare R2 Storage}.
\end{itemize}
\end{frame}
